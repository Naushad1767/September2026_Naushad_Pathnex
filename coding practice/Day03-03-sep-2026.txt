# ansible - Install apache and Start Service

- name: Install apache
 yum:
   name: httpd
   start: present

- name: start apache
  service:
    name: httpd
    state: started
    enabled: yes

#Kubernetes - Create a Replicaset With 3 Replicas

apiVersion: apps/v1
kind: ReplicaSet
metadta:
  name: pathnex-replicaset
spec:
  replicas: 3
  selector:
    matchLables:
      app: pathnex-app
  template:
    metadata:
      labels:
        app: pathnex-app
    spec:
      conatiners:
        - name: app
          image: nginx
          ports:
            - containerPort: 80
