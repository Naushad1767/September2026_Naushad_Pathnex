#Day 09 — Docker Integration with Jenkins & GitLab
#Ansible — Setup Docker Container for Nginx
- name: Setup Docker Container for Nginx
  hosts: all
  become: yes
  tasks:
    - name: Pull Nginx Image
      docker_image:
        name: Nginx
        source: Pull
    - name: Run Nginx Container
      docker_container:
        name: nginx-container
        image: nginx
        state: started
        published_ports:
          - "8080:80"

# Kubernetes — Horizontal Pod Autoscaler (HPA)
apiVersion: apps/v1
kind: Deployment
metadata:
  name: pathnex-deployment

spec:
  replicas: 1
  selector:
    matchLabls:
      app: pathnex-app
  
  template:
    metadata:
      labels:
        app: pathnex-app

  spec:
    Containers:
      - name: nginx
        image: nginx
        ports: 
          - containerPort: 80

-----

apiVersion: Autoscaling/v2
kind: HorizontalPodAutoscaler
metadata:
  name: pathnex-hpa

spec:
scalerTargetRef:
  apiVersion: apps/v1
  kind: Deployment
  name: pathnex-deployment
  minReplicas: 1
  maxReplicas: 10
  targetCPUUtilizationPercentage: 50
